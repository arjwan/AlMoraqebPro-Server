// دالة التحقق من تسجيل الدخول عبر مطابقة البيانات الموجودة في النظام المركزى (create-company) function handleLoginMatch(inputCompanyCode, inputPhone, inputUsername, inputPassword) { // جلب قائمة الشركات المسجلة في النظام (المحفوظة من صفحة create-company.html) const registeredCompanies = JSON.parse(localStorage.getItem('registeredCompanies')) || []; // البحث عن مطابقة صحيحة const matchedCompany = registeredCompanies.find(comp => { // شرط حالة الشركة يجب أن يكون مفعل (active أو ليس متوقفاً) const isActive = comp.status !== 'stopped'; // مطابقة (رمز الشركة ورقم الهاتف) أو (اسم المستخدم وكلمة المرور) const isPhoneMatch = (comp.id === inputCompanyCode || comp.company_id === inputCompanyCode) && comp.phone === inputPhone; const isCredentialMatch = comp.username === inputUsername && comp.password === inputPassword; return isActive && (isPhoneMatch || isCredentialMatch); }); if (matchedCompany) { const companyCode = matchedCompany.id || matchedCompany.company_id; // تخزين معرف الشركة النشطة الحالية localStorage.setItem('activeCompanyId', companyCode); localStorage.setItem('activeCompanyName', matchedCompany.nameAr || matchedCompany.name_ar); // رسالة النجاح والتوجيه الفوري للوحة التحكم alert(`✨ مرحباً بك ${matchedCompany.manager} في نظام ${matchedCompany.nameAr}`); window.location.href = `admin.html?company=${encodeURIComponent(companyCode)}`; return true; } else { alert('❌ البيانات المدخلة غير مطابقة، أو أن حساب الشركة متوقف مؤقتاً من قبل المطور.'); return false; } }
